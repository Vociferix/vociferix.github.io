234macro_rules! def_enum_prop {
235    ($(#[$($attr:tt)*])* enum $name:ident { $($(#[$($vattr:tt)*])* $var:ident => $val:path),* $(,)* } => $map:expr , default = $dvar:ident) => {
236        $(#[$($attr)*])*
237        #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
238        pub enum $name {
239            $(
240                $(#[$($vattr)*])*
241                $var,
242            )*
243        }
244
245        impl $name {
246            /// Gets the property value of a code point.
247            pub fn of(ch: char) -> Self {
248                let val = const { &($map) }.get(ch);
249                match val {
250                    $(
251                        $val => Self::$var,
252                    )*
253                    #[allow(unreachable_patterns)]
254                    _ => Self::$dvar,
255                }
256            }
257        }
258
259        impl Property for $name {
260            fn contains(self, ch: char) -> bool {
261                self == Self::of(ch)
262            }
263        }
264
265        impl core::ops::Not for $name {
266            type Output = Not<Self>;
267
268            fn not(self) -> Not<Self> {
269                Not(self)
270            }
271        }
272
273        impl<R: Property> core::ops::BitAnd<R> for $name {
274            type Output = And<Self, R>;
275
276            fn bitand(self, rhs: R) -> And<Self, R> {
277                And(self, rhs)
278            }
279        }
280
281        impl<R: Property> core::ops::BitOr<R> for $name {
282            type Output = Or<Self, R>;
283
284            fn bitor(self, rhs: R) -> Or<Self, R> {
285                Or(self, rhs)
286            }
287        }
288    };
289}

906pub enum IndicConjunctBreak {
907    None,
908    Linker,
909    Consonant,
910    Extend,
911}
912
913impl IndicConjunctBreak {
914    /// Gets the property value of a code point.
915    pub fn of(ch: char) -> Self {
916        if let Script::Bengali
917        | Script::Devanagari
918        | Script::Gujarati
919        | Script::Malayalam
920        | Script::Oriya
921        | Script::Telugu = Script::of(ch)
922        {
923            match IndicSyllabicCategory::of(ch) {
924                IndicSyllabicCategory::Virama => Self::Linker,
925                IndicSyllabicCategory::Consonant => Self::Consonant,
926                _ => Self::None,
927            }
928        } else if let GraphemeClusterBreak::ZWJ | GraphemeClusterBreak::Extend =
929            GraphemeClusterBreak::of(ch)
930        {
931            if ch != '\u{200c}' {
932                Self::Extend
933            } else {
934                Self::None
935            }
936        } else {
937            Self::None
938        }
939    }
940}
